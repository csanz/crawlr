import * as THREE from 'three';
import * as RAPIER from '@dimforge/rapier3d';
import { loadTexture } from './TextureLoader.js';

/**
 * Creates the visual ground plane and its physics collider.
 * @param {THREE.Scene} scene - The Three.js scene.
 * @param {RAPIER.World} world - The Rapier physics world.
 * @param {THREE.WebGLRenderer} renderer - The Three.js renderer (for texture loading).
 * @returns {THREE.Mesh} The ground plane mesh.
 */
export function createGround(scene, world, renderer) {
    // Load the ground texture
    const groundTexture = loadTexture('public/assets/ground-texture-2.png', renderer, 30, 30); 

    // Add a flat ground visuals
    const groundGeometry = new THREE.PlaneGeometry(50, 50); // Size matches physics collider
    const groundMaterial = new THREE.MeshStandardMaterial({ 
        map: groundTexture,
        side: THREE.DoubleSide 
    }); 
    const groundPlane = new THREE.Mesh(groundGeometry, groundMaterial);
    groundPlane.rotation.x = Math.PI / 2;
    groundPlane.receiveShadow = true; // Allow ground to receive shadows
    scene.add(groundPlane);

    // Create the ground in the physics world
    const groundColliderDesc = RAPIER.ColliderDesc.cuboid(25.0, 0.1, 25.0); // Matches visual plane size / 2
    // Set the ground collider's position so its top surface is at y=0
    groundColliderDesc.setTranslation(0.0, -0.1, 0.0);
    world.createCollider(groundColliderDesc);

    console.log("Ground created.");
    return groundPlane;
}

    run: 0 // 1 for run, 0 otherwise

/**
 * Initializes the keyboard event listeners for player movement.
 * @param {RAPIER.RigidBody} playerBody - The player's Rapier rigid body instance.
 */
export function initInputHandler(playerBody) {
    // Key press event listeners
    window.addEventListener('keydown', (event) => {
        switch (event.key.toLowerCase()) { // Use toLowerCase for consistency
            case 'w':
                moveState.forward = 1;
                break;
            case 's':
                moveState.backward = 1;
                break;
            case 'a':
                moveState.left = 1;
                break;
            case 'd':
                moveState.right = 1;
                break;
            case 'shift':
                moveState.run = 1;
                break;
            case ' ': // Space bar for jump
                if (playerBody) { // Ensure playerBody is available
                    const currentVel = playerBody.linvel();
                    if (Math.abs(currentVel.y) < 0.1) { // Simple ground check approximation
                        playerBody.applyImpulse({ x: 0, y: jumpForce, z: 0 }, true);
                    }
                }
                break;
        }
    });

    window.addEventListener('keyup', (event) => {
        switch (event.key.toLowerCase()) { // Use toLowerCase for consistency
            case 'w':
                moveState.forward = 0;
                break;
            case 's':
                moveState.backward = 0;
                break;
            case 'a':
                moveState.left = 0;
                break;
            case 'd':
                moveState.right = 0;
                break;
            case 'shift':
                moveState.run = 0;
                break;
        }
    });

    console.log("Input handler initialized.");
}

import * as THREE from 'three';
import * as RAPIER from '@dimforge/rapier3d';
import { loadTexture } from './TextureLoader.js';
import { moveState } from './InputHandler.js'; // Import moveState to use in updatePlayerMovement

// Player constants
const playerSpeed = 5.0;
const playerRunMultiplier = 2.0;

// Reusable vectors for movement calculation
const cameraForward = new THREE.Vector3();
const cameraRight = new THREE.Vector3();
const desiredVelocity = new THREE.Vector3();
const worldUp = new THREE.Vector3(0, 1, 0);

/**
 * Creates the player's visual mesh and physics body.
 * @param {THREE.Scene} scene - The Three.js scene.
 * @param {RAPIER.World} world - The Rapier physics world.
 * @param {THREE.WebGLRenderer} renderer - The Three.js renderer (for texture loading).
 * @returns {{playerMesh: THREE.Mesh, playerBody: RAPIER.RigidBody}} 
 */
export function createPlayer(scene, world, renderer) {
    // Load the player texture
    const playerTexture = loadTexture('public/assets/player-texture.png', renderer);

    // Create visual representation for the player
    const playerGeometry = new THREE.BoxGeometry(1, 1, 1); // Match Rapier collider size (0.5 * 2)
    const playerMaterial = new THREE.MeshStandardMaterial({ 
        map: playerTexture // Apply the texture
    }); 
    const playerMesh = new THREE.Mesh(playerGeometry, playerMaterial);
    playerMesh.castShadow = true; // Allow player to cast shadows
    scene.add(playerMesh);

    // Create a player object with physics properties
    const playerBodyDesc = RAPIER.RigidBodyDesc.dynamic().setTranslation(0.0, 1.0, 0.0);
    const playerBody = world.createRigidBody(playerBodyDesc);
    const playerColliderDesc = RAPIER.ColliderDesc.cuboid(0.5, 0.5, 0.5);
    world.createCollider(playerColliderDesc, playerBody);

    console.log("Player created.");
    return { playerMesh, playerBody };
}

/**
 * Updates the player's physics body velocity based on input and camera direction.
 * @param {RAPIER.RigidBody} playerBody - The player's rigid body.
 * @param {THREE.Camera} camera - The scene camera.
 */
export function updatePlayerMovement(playerBody, camera) {
    // Get camera direction vectors
    camera.getWorldDirection(cameraForward);
    // Get right vector using cross product (Corrected order)
    cameraRight.crossVectors(cameraForward, worldUp).normalize(); 

    // Project onto XZ plane
    cameraForward.y = 0;
    cameraRight.y = 0;
    cameraForward.normalize();
    cameraRight.normalize();

    // Calculate direction based on imported moveState
    let moveDirectionZ = moveState.forward - moveState.backward;
    let moveDirectionX = moveState.right - moveState.left;
    
    // Combine directions based on camera orientation
    desiredVelocity.set(0, 0, 0); // Reset desired velocity
    desiredVelocity.addScaledVector(cameraForward, moveDirectionZ);
    desiredVelocity.addScaledVector(cameraRight, moveDirectionX);
    
    // Normalize only if there is movement input to avoid NaN issues
    if (moveDirectionZ !== 0 || moveDirectionX !== 0) {
        desiredVelocity.normalize();
    }

    // Apply speed and run multiplier
    const currentSpeed = moveState.run ? playerSpeed * playerRunMultiplier : playerSpeed;
    desiredVelocity.multiplyScalar(currentSpeed);

    // Set the rigid body's velocity
    const currentVelocity = playerBody.linvel();
    playerBody.setLinvel({ 
        x: desiredVelocity.x, 
        y: currentVelocity.y, // Preserve Y velocity for gravity/jump
        z: desiredVelocity.z 
    }, true); // Wake up the body
}

import * as THREE from 'three';
import * as RAPIER from '@dimforge/rapier3d';
import { loadTexture } from './TextureLoader.js';

/**
 * Creates the pushable block's visual mesh and physics body.
 * @param {THREE.Scene} scene - The Three.js scene.
 * @param {RAPIER.World} world - The Rapier physics world.
 * @param {THREE.WebGLRenderer} renderer - The Three.js renderer (for texture loading).
 * @param {THREE.Vector3} initialPosition - The starting position for the block.
 * @returns {{blockMesh: THREE.Mesh, blockBody: RAPIER.RigidBody}}
 */
export function createPushableBlock(scene, world, renderer, initialPosition = new THREE.Vector3(2, 0.5, 0)) {
    // Load the block texture
    const blockTexture = loadTexture('public/assets/block-texture.png', renderer);

    // Create visual representation for the block
    const blockGeometry = new THREE.BoxGeometry(1, 1, 1);
    const blockMaterial = new THREE.MeshStandardMaterial({ 
        map: blockTexture // Apply the texture
    });
    const blockMesh = new THREE.Mesh(blockGeometry, blockMaterial);
    blockMesh.castShadow = true; // Allow block to cast shadows
    blockMesh.position.copy(initialPosition); // Set initial visual position
    scene.add(blockMesh);

    // Create pushable block physics body and collider
    const blockBodyDesc = RAPIER.RigidBodyDesc.dynamic().setTranslation(initialPosition.x, initialPosition.y, initialPosition.z);
    const blockBody = world.createRigidBody(blockBodyDesc);
    const blockColliderDesc = RAPIER.ColliderDesc.cuboid(0.5, 0.5, 0.5);
    world.createCollider(blockColliderDesc, blockBody);
    
    console.log("Pushable block created.");
    return { blockMesh, blockBody };
}

import * as THREE from 'three';

/**
 * Loads a texture from the given URL.
 * Configures repeat wrapping suitable for ground textures.
 *
 * @param {string} url - The path to the texture image.
 * @param {THREE.WebGLRenderer} renderer - The Three.js renderer instance.
 * @param {number} [repeatX=1] - How many times the texture should repeat horizontally.
 * @param {number} [repeatY=1] - How many times the texture should repeat vertically.
 * @returns {THREE.Texture} The configured texture object.
 */
export function loadTexture(url, renderer, repeatX = 1, repeatY = 1) {
    const loader = new THREE.TextureLoader();
    const texture = loader.load(url);

    // Set wrapping to repeat
    texture.wrapS = THREE.RepeatWrapping;
    texture.wrapT = THREE.RepeatWrapping;

    // Set repetition amount
    texture.repeat.set(repeatX, repeatY);

    // Optional: Improve texture quality
    texture.magFilter = THREE.LinearFilter;
    texture.minFilter = THREE.LinearMipmapLinearFilter;
    // Use the passed renderer instance
    texture.anisotropy = Math.min(16, renderer.capabilities.getMaxAnisotropy());

    return texture;
}
